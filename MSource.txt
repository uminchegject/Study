node = hou.pwd()
geo = node.geometry()

path = "C:/Users/yusuke.nakamura/Desktop/test.txt"

with open(path, "w", encoding="utf-8") as f:

    # =========================================
    # POINT ATTRIBUTES
    # =========================================

    f.write("=== POINT ATTRIBUTES ===\n\n")

    point_attribs = geo.pointAttribs()

    for pt in geo.points():

        f.write("Point {}\n".format(pt.number()))

        for attrib in point_attribs:

            name = attrib.name()
            value = pt.attribValue(attrib)

            f.write("  {} : {}\n".format(name, value))

        f.write("\n")

    # =========================================
    # PRIMITIVE ATTRIBUTES
    # =========================================

    f.write("\n=== PRIMITIVE ATTRIBUTES ===\n\n")

    prim_attribs = geo.primAttribs()

    for prim in geo.prims():

        f.write("Prim {}\n".format(prim.number()))

        for attrib in prim_attribs:

            name = attrib.name()
            value = prim.attribValue(attrib)

            f.write("  {} : {}\n".format(name, value))

        f.write("\n")

    # =========================================
    # VERTEX ATTRIBUTES
    # =========================================

    f.write("\n=== VERTEX ATTRIBUTES ===\n\n")

    vertex_attribs = geo.vertexAttribs()

    for prim in geo.prims():

        for vtx in prim.vertices():

            f.write(
                "Vertex prim={} vertex={}\n".format(
                    prim.number(),
                    vtx.number()
                )
            )

            for attrib in vertex_attribs:

                name = attrib.name()
                value = vtx.attribValue(attrib)

                f.write("  {} : {}\n".format(name, value))

            f.write("\n")

    # =========================================
    # DETAIL ATTRIBUTES
    # =========================================

    f.write("\n=== DETAIL ATTRIBUTES ===\n\n")

    detail_attribs = geo.globalAttribs()

    for attrib in detail_attribs:

        name = attrib.name()
        value = geo.attribValue(attrib)

        f.write("{} : {}\n".format(name, value))

print("Exported:", path)
